  <script>
    // -------------------------------------------------------------
    // DÁN ĐƯỜNG LINK CSV XUẤT BẢN TỪ GOOGLE SHEETS CỦA BẠN VÀO ĐÂY:
    // -------------------------------------------------------------
    const SHEET_CSV_URL = "DÁN_LINK_CSV_CỦA_BẠN_VÀO_ĐÂY";

    let candidateMap = new Map(); // Lưu dữ liệu dạng: Mã số => Thông tin
    let html5QrCode = null;
    let isScanning = false;
    let lastScannedCode = null;

    // Tải dữ liệu từ Google Sheets
    async function loadDataFromSheets() {
      const statusEl = document.getElementById("dbStatus");
      if (!SHEET_CSV_URL || SHEET_CSV_URL.includes("DÁN_LINK_CSV")) {
        statusEl.innerHTML = "<b style='color:red;'>Chưa cấu hình URL Google Sheets!</b>";
        return;
      }

      statusEl.textContent = "Đang đồng bộ dữ liệu từ Google Sheets...";
      
      // Thêm timestamp để tránh cache trình duyệt khi bạn sửa sheet
      const urlWithCacheBuster = SHEET_CSV_URL + (SHEET_CSV_URL.includes('?') ? '&' : '?') + 't=' + Date.now();

      Papa.parse(urlWithCacheBuster, {
        download: true,
        header: true,
        skipEmptyLines: true,
        complete: function(results) {
          candidateMap.clear();
          results.data.forEach(row => {
            // Lấy mã số làm khóa tra cứu (loại bỏ khoảng trắng thừa)
            const id = (row['Mã số'] || row['ID'] || row['MaSo'] || '').trim();
            if (id) {
              candidateMap.set(id.toLowerCase(), {
                id: id,
                name: row['Họ và tên'] || row['HoTen'] || 'Chưa cập nhật',
                room: row['Phòng thi'] || row['PhongThi'] || 'Chưa xếp phòng',
                field: row['Lĩnh vực thi'] || row['LinhVuc'] || 'Chưa xếp lĩnh vực'
              });
            }
          });
          statusEl.textContent = `Đã đồng bộ ${candidateMap.size} thí sinh từ Sheets`;
          statusEl.style.color = "#16a34a";
        },
        error: function(err) {
          statusEl.textContent = "Lỗi khi tải dữ liệu Sheets. Kiểm tra lại quyền chia sẻ!";
          statusEl.style.color = "#dc2626";
        }
      });
    }

    // Khi quét mã thành công
    function onScanSuccess(decodedText, decodedResult) {
      const code = decodedText.trim();
      
      // Tránh quét liên tục cùng 1 mã trong thời gian quá ngắn (1.5 giây)
      if (code === lastScannedCode) return;
      lastScannedCode = code;
      setTimeout(() => { lastScannedCode = null; }, 1500);

      if (navigator.vibrate) {
        navigator.vibrate(100);
      }

      const card = document.getElementById("resultCard");
      const header = document.getElementById("resHeader");
      const rawEl = document.getElementById("resRaw");

      rawEl.textContent = `Mã quét được: ${code}`;
      card.classList.add("active");

      // Tra cứu trong Map (không phân biệt chữ hoa/thường)
      const found = candidateMap.get(code.toLowerCase());

      if (found) {
        card.classList.remove("not-found");
        header.innerHTML = "<span>✓</span> Đã tìm thấy thí sinh";
        document.getElementById("resId").textContent = found.id;
        document.getElementById("resName").textContent = found.name;
        document.getElementById("resRoom").textContent = found.room;
        document.getElementById("resField").textContent = found.field;
        document.getElementById("dataFields").style.display = "block";
      } else {
        card.classList.add("not-found");
        header.innerHTML = "<span>✕</span> Không tìm thấy thí sinh!";
        document.getElementById("resId").textContent = code;
        document.getElementById("resName").textContent = "Không có trong danh sách";
        document.getElementById("resRoom").textContent = "--";
        document.getElementById("resField").textContent = "--";
        document.getElementById("dataFields").style.display = "block";
      }
    }

    async function startScanner() {
      if (!html5QrCode) {
        html5QrCode = new Html5Qrcode("reader");
      }

      const config = {
        fps: 15,
        // Khung quét chữ nhật tối ưu cho quét Barcode mã vạch ngang
        qrbox: { width: 300, height: 160 },
        aspectRatio: 1.333334
      };

      try {
        await html5QrCode.start(
          { facingMode: "environment" },
          config,
          onScanSuccess
        );
        document.getElementById("btnStart").disabled = true;
        document.getElementById("btnStop").disabled = false;
        isScanning = true;
      } catch (err) {
        alert("Không thể khởi động camera: " + err);
      }
    }

    async function stopScanner() {
      if (html5QrCode && isScanning) {
        await html5QrCode.stop();
        document.getElementById("btnStart").disabled = false;
        document.getElementById("btnStop").disabled = true;
        isScanning = false;
      }
    }

    // Tự động tải dữ liệu Google Sheets khi vừa mở trang
    window.addEventListener("DOMContentLoaded", loadDataFromSheets);
  </script>
